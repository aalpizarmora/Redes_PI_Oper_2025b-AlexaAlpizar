Configuración de switch L3:

Switch>enable
Switch#configure terminal

Switch(config)#ip routing
Switch(config)#vlan 620
Switch(config-vlan)#name Isla2
Switch(config-vlan)#exit
Switch(config)#vlan 630
Switch(config-vlan)#name Isla3
Switch(config-vlan)#exit

Switch(config)#interface vlan 620
Switch(config-if)#
Switch(config-if)#ip address 172.16.123.33 255.255.255.240
Switch(config-if)#no shutdown

Switch(config-if)#interface vlan 630
Switch(config-if)#
%LINK-5-CHANGED: Interface Vlan630, changed state to up
Switch(config-if)#ip address 172.16.123.49 255.255.255.240
Switch(config-if)#no shutdown

DHCP Pool L2 (Isla 2):
Switch(config-if)#ip dhcp excluded-address 172.16.123.33 172.16.123.34
Switch(config)#ip dhcp pool VLAN620
Switch(dhcp-config)#network 172.16.123.32 255.255.255.240
Switch(dhcp-config)#default-router 172.16.123.33
Switch(dhcp-config)#dns-server 8.8.8.8
Switch(dhcp-config)#exit

DHCP Pool L2 (Isla 3):
Switch(config)#ip dhcp excluded-address 172.16.123.49 172.16.123.50
Switch(config)#ip dhcp pool VLAN630
Switch(dhcp-config)#network 172.16.123.48 255.255.255.240
Switch(dhcp-config)#default-router 172.16.123.49
Switch(dhcp-config)#dns-server 8.8.8.8
Switch(dhcp-config)#exit

Configuración final L3:
Switch#enable
Switch#configure terminal

Switch(config)#interface FastEthernet0/2
Switch(config-if)#switchport trunk encapsulation dot1q
Switch(config-if)#switchport mode trunk
Switch(config-if)#switchport trunk allowed vlan 620
Switch(config-if)#description Trunk_L2_Isla2
Switch(config-if)#exit

Switch(config)#interface FastEthernet0/3
Switch(config-if)#switchport trunk encapsulation dot1q
Switch(config-if)#switchport mode trunk
Switch(config-if)#switchport trunk allowed vlan 630
Switch(config-if)#description Trunk_L2_Isla3
Switch(config-if)#exit
